```python
"""Kernel interface + registry.

A `Kernel` is a pure function of (world_seed, world_xz, params) that
returns a heightmap value (meters) and per-biome weight values
(0..1, dimensionless). Subclasses MUST define class attribute `kind`
(matches biome_catalog.json's `generator.kernel` field) and implement
`height` + `biome_weight`.

All kernel methods must be pure: same inputs -> same outputs. No I/O,
no mutable global state, no time-dependent behavior.
"""
from __future__ import annotations

from abc import ABC, abstractmethod
from typing import Type


class KernelError(RuntimeError):
    """Raised for kernel-system misuse (unknown kernel, registry clashes)."""


class Kernel(ABC):
    """Abstract base for terrain kernels.

    Subclasses must set the class attribute `kind` to a unique short
    string (e.g. "noise_stack") and implement `height()` +
    `biome_weight()`. The `kind` value is what `biome_catalog.json`'s
    `generator.kernel` field references.
    """

    kind: str = ""  # subclasses override

    @abstractmethod
    def height(self, world_x: float, world_z: float, world_seed: int,
               params: dict) -> float:
        """Return the terrain height (meters) at world position
        (world_x, world_z) for the given kernel params + world seed."""

    @abstractmethod
    def biome_weight(self, biome_name: str, world_x: float, world_z: float,
                     world_seed: int, params: dict) -> float:
        """Return the unnormalized biome weight (0..1, dimensionless)
        for `biome_name` at world position (world_x, world_z). The
        composer normalizes across biomes — this method just returns
        the raw weight."""


class KernelRegistry:
    """Registry of kernel kinds to instances.

    Each registered subclass is instantiated once and held as a
    singleton. Subclasses must be stateless (kernels are pure functions
    of their args; instance state would break the purity contract).
    """

    def __init__(self) -> None:
        self._kernels: dict[str, Kernel] = {}

    def register(self, kernel_cls: Type[Kernel]) -> None:
        kind = kernel_cls.kind
        if not kind:
            raise KernelError(
                f"kernel class {kernel_cls.__name__} must set class attr `kind`")
        if kind in self._kernels:
            raise KernelError(f"kernel {kind!r} already registered")
        self._kernels[kind] = kernel_cls()

    def get(self, kind: str) -> Kernel:
        if kind not in self._kernels:
            raise KernelError(f"unknown kernel {kind!r}")
        return self._kernels[kind]

    def all_kinds(self) -> list[str]:
        return sorted(self._kernels.keys())
```

Create `pipeline/kernels/noise_stack.py`:

```python
"""Noise-stack kernel — multi-octave gradient noise.

The math:
- For each octave i in 0..octaves-1:
    freq = base_frequency_per_m * lacunarity^i
    amp = persistence^i
    n_i = gradient_noise_2d(world_x * freq, world_z * freq, seed)
- height_signed = sum(n_i * amp) / sum(amp)        # in -1..1
- height = elevation_base_m + height_signed * elevation_amplitude_m

`gradient_noise_2d` is a hand-rolled deterministic hash + gradient
look-up implementation that the GDScript port can mirror bit-for-bit
(modulo float precision). See `_gradient_noise_2d` for the exact
operations.

The seed used for `_gradient_noise_2d` is `world_seed + seed_offset`
(from params). This lets two biomes using NoiseStackKernel with the
same world_seed produce different patterns by changing seed_offset.
"""
from __future__ import annotations

import math

from kernels.base import Kernel


REQUIRED_PARAMS = (
    "elevation_base_m",
    "elevation_amplitude_m",
    "octaves",
    "lacunarity",
    "persistence",
    "base_frequency_per_m",
    "seed_offset",
)


def _hash2(ix: int, iz: int, seed: int) -> int:
    """Deterministic integer hash of (ix, iz, seed) -> 32-bit int.

    Uses the same magic constants and mix steps as the GDScript port
    will so the two implementations agree.
    """
    h = (seed * 374761393) & 0xFFFFFFFF
    h = (h + ix * 668265263) & 0xFFFFFFFF
    h = (h + iz * 2147483647) & 0xFFFFFFFF
    h = h ^ (h >> 13)
    h = (h * 1274126177) & 0xFFFFFFFF
    h = h ^ (h >> 16)
    return h


def _gradient(ix: int, iz: int, seed: int) -> tuple[float, float]:
    """Return a deterministic 2D unit vector for grid cell (ix, iz)."""
    h = _hash2(ix, iz, seed)
    angle = (h / 0xFFFFFFFF) * 2.0 * math.pi
    return (math.cos(angle), math.sin(angle))


def _fade(t: float) -> float:
    """Quintic smoothstep used by Perlin: 6t^5 - 15t^4 + 10t^3."""
    return t * t * t * (t * (t * 6.0 - 15.0) + 10.0)


def _gradient_noise_2d(x: float, z: float, seed: int) -> float:
    """Perlin-style gradient noise at continuous (x, z) for given seed.

    Output is in approximately [-1, 1]. Continuous + differentiable.
    """
    ix = int(math.floor(x))
    iz = int(math.floor(z))
    fx = x - ix
    fz = z - iz

    # Gradients at the 4 corners of the unit cell.
    g00 = _gradient(ix,     iz,     seed)
    g10 = _gradient(ix + 1, iz,     seed)
    g01 = _gradient(ix,     iz + 1, seed)
    g11 = _gradient(ix + 1, iz + 1, seed)

    # Distance vectors from each corner to the sample point.
    d00 = (fx,       fz)
    d10 = (fx - 1.0, fz)
    d01 = (fx,       fz - 1.0)
    d11 = (fx - 1.0, fz - 1.0)

    # Dot products.
    n00 = g00[0] * d00[0] + g00[1] * d00[1]
    n10 = g10[0] * d10[0] + g10[1] * d10[1]
    n01 = g01[0] * d01[0] + g01[1] * d01[1]
    n11 = g11[0] * d11[0] + g11[1] * d11[1]

    # Smooth interpolation in x then z.
    u = _fade(fx)
    v = _fade(fz)
    nx0 = n00 * (1.0 - u) + n10 * u
    nx1 = n01 * (1.0 - u) + n11 * u
    return nx0 * (1.0 - v) + nx1 * v


class NoiseStackKernel(Kernel):
    """Multi-octave fBm over `_gradient_noise_2d`."""

    kind = "noise_stack"

    def _validate(self, params: dict) -> None:
        missing = [p for p in REQUIRED_PARAMS if p not in params]
        if missing:
            raise KeyError(f"NoiseStackKernel missing params: {missing!r}")

    def height(self, world_x: float, world_z: float, world_seed: int,
               params: dict) -> float:
        self._validate(params)
        seed = int(world_seed) + int(params["seed_offset"])
        octaves = int(params["octaves"])
        lacunarity = float(params["lacunarity"])
        persistence = float(params["persistence"])
        base_freq = float(params["base_frequency_per_m"])
        base_m = float(params["elevation_base_m"])
        amp_m = float(params["elevation_amplitude_m"])

        total = 0.0
        amp = 1.0
        freq = base_freq
        norm = 0.0
        for _ in range(octaves):
            total += _gradient_noise_2d(world_x * freq, world_z * freq, seed) * amp
            norm += amp
            amp *= persistence
            freq *= lacunarity
        if norm > 0.0:
            total /= norm
        return base_m + total * amp_m

    def biome_weight(self, biome_name: str, world_x: float, world_z: float,
                     world_seed: int, params: dict) -> float:
        # NoiseStackKernel doesn't itself drive biome assignment — the
        # composer's biome map (a separate noise field, see
        # kernel_composer.py) determines biome boundaries. Future
        # kernels (e.g. ErosionBakedKernel) may return slope- or
        # height-derived weights to bias biome placement.
        return 1.0
```

Create `pipeline/kernel_composer.py`:

```python
"""KernelComposer — combines per-biome kernels into world-level functions.

Owns:
- the biome -> (kernel, params) mapping (parsed from biome_catalog.json)
- the biome-weight field (a per-biome noise function over world XZ)

API:
    sample_height(world_x, world_z, world_seed) -> float
    sample_biome_weights(world_x, world_z, world_seed) -> dict[biome, float]

The biome-weight field is a softmax over per-biome scalar fields:
each biome has a scalar field B_i(x, z) = gradient_noise_2d(x / scale,
z / scale, seed + i*1000). The weight for biome i at (x, z) is
  exp(B_i / temperature) / sum_j(exp(B_j / temperature))

This gives each biome a "territory" of approximate scale `biome_scale_m`
that varies smoothly. Temperature controls boundary sharpness;
v1 uses 0.5 (moderately sharp, smooth at boundaries).

Height at (x, z) is the weighted sum of each biome's kernel.height()
output, weighted by the biome weights. This produces a smooth
blend at biome boundaries — at a 50/50 alpine/desert boundary the
height is halfway between alpine's noise and desert's noise.
"""
from __future__ import annotations

import math

from kernels.base import Kernel, KernelRegistry
from kernels.noise_stack import _gradient_noise_2d


class ComposerError(RuntimeError):
    """Raised for composer-level catalog issues."""


# Per-biome biome-map noise frequency. Each biome's territory map uses
# a different seed offset (biome index * BIOME_SEED_STRIDE).
BIOME_SEED_STRIDE = 1000

# Temperature for softmax over biome scalar fields. Lower = sharper
# biome boundaries; higher = more gradient zones.
DEFAULT_BIOME_TEMPERATURE = 0.5


class _BiomeBinding:
    __slots__ = ("name", "kernel", "params", "biome_index")

    def __init__(self, name: str, kernel: Kernel, params: dict, biome_index: int):
        self.name = name
        self.kernel = kernel
        self.params = params
        self.biome_index = biome_index


class KernelComposer:
    """Combines per-biome kernels at world level."""

    def __init__(self, catalog: dict, registry: KernelRegistry,
                 biome_temperature: float = DEFAULT_BIOME_TEMPERATURE):
        self._biomes: list[_BiomeBinding] = []
        self._biome_scale_m = float(catalog.get("biome_scale_m", 1024.0))
        self._temperature = biome_temperature
        for i, b in enumerate(catalog["biomes"]):
            if "generator" not in b:
                raise ComposerError(f"biome {b['name']!r}: missing generator block")
            gen = b["generator"]
            kind = gen.get("kernel")
            if kind is None:
                raise ComposerError(f"biome {b['name']!r}: generator.kernel missing")
            try:
                kernel = registry.get(kind)
            except Exception as e:
                raise ComposerError(
                    f"biome {b['name']!r}: kernel {kind!r}: {e}") from e
            params = gen.get("params", {})
            self._biomes.append(_BiomeBinding(b["name"], kernel, params, i))

    def biome_names(self) -> list[str]:
        return [b.name for b in self._biomes]

    def _biome_scalar(self, biome_index: int, world_x: float, world_z: float,
                      world_seed: int) -> float:
        """Per-biome scalar field used for biome assignment. Each biome
        gets its own noise pattern over world XZ; softmax over these
        gives smooth biome boundaries."""
        seed = world_seed + biome_index * BIOME_SEED_STRIDE
        u = world_x / self._biome_scale_m
        v = world_z / self._biome_scale_m
        return _gradient_noise_2d(u, v, seed)

    def sample_biome_weights(self, world_x: float, world_z: float,
                             world_seed: int) -> dict[str, float]:
        # Softmax: stable form (subtract max before exp).
        scalars = [
            self._biome_scalar(b.biome_index, world_x, world_z, world_seed)
            for b in self._biomes
        ]
        mx = max(scalars)
        t = max(self._temperature, 1e-6)
        exps = [math.exp((s - mx) / t) for s in scalars]
        total = sum(exps)
        return {
            b.name: (e / total) for b, e in zip(self._biomes, exps)
        }

    def sample_height(self, world_x: float, world_z: float,
                      world_seed: int) -> float:
        weights = self.sample_biome_weights(world_x, world_z, world_seed)
        h = 0.0
        for b in self._biomes:
            w = weights[b.name]
            if w < 1e-6:
                continue
            h += w * b.kernel.height(world_x, world_z, world_seed, b.params)
        return h
```

```gdscript
class_name NoiseStackKernel
extends Kernel

# Multi-octave fBm over gradient noise. MUST agree numerically with
# pipeline/kernels/noise_stack.py — the hash function, gradient
# computation, fade curve, and accumulation order are all pinned to
# match. The cross-impl test (tests/test_kernel_cross_impl.py +
# scripts/kernels/KernelDump.gd) verifies max delta < epsilon.

const REQUIRED_PARAMS := [
	"elevation_base_m", "elevation_amplitude_m",
	"octaves", "lacunarity", "persistence",
	"base_frequency_per_m", "seed_offset",
]


func _init() -> void:
	kind = "noise_stack"


# 32-bit unsigned hash. Mirrors Python _hash2 exactly. We use ints
# masked to 32 bits because GDScript ints are 64-bit, and Python's
# `& 0xFFFFFFFF` keeps them 32-bit.
static func _hash2(ix: int, iz: int, seed: int) -> int:
	var h: int = (seed * 374761393) & 0xFFFFFFFF
	h = (h + ix * 668265263) & 0xFFFFFFFF
	h = (h + iz * 2147483647) & 0xFFFFFFFF
	h = h ^ (h >> 13)
	h = (h * 1274126177) & 0xFFFFFFFF
	h = h ^ (h >> 16)
	return h


static func _gradient(ix: int, iz: int, seed: int) -> Vector2:
	var h: int = _hash2(ix, iz, seed)
	var angle: float = (float(h) / float(0xFFFFFFFF)) * TAU
	return Vector2(cos(angle), sin(angle))


static func _fade(t: float) -> float:
	return t * t * t * (t * (t * 6.0 - 15.0) + 10.0)


static func _gradient_noise_2d(x: float, z: float, seed: int) -> float:
	var ix: int = int(floor(x))
	var iz: int = int(floor(z))
	var fx: float = x - float(ix)
	var fz: float = z - float(iz)

	var g00 := _gradient(ix,     iz,     seed)
	var g10 := _gradient(ix + 1, iz,     seed)
	var g01 := _gradient(ix,     iz + 1, seed)
	var g11 := _gradient(ix + 1, iz + 1, seed)

	var d00 := Vector2(fx,       fz)
	var d10 := Vector2(fx - 1.0, fz)
	var d01 := Vector2(fx,       fz - 1.0)
	var d11 := Vector2(fx - 1.0, fz - 1.0)

	var n00: float = g00.dot(d00)
	var n10: float = g10.dot(d10)
	var n01: float = g01.dot(d01)
	var n11: float = g11.dot(d11)

	var u: float = _fade(fx)
	var v: float = _fade(fz)
	var nx0: float = n00 * (1.0 - u) + n10 * u
	var nx1: float = n01 * (1.0 - u) + n11 * u
	return nx0 * (1.0 - v) + nx1 * v


func height(world_x: float, world_z: float, world_seed: int,
            params: Dictionary) -> float:
	for p in REQUIRED_PARAMS:
		if not params.has(p):
			push_error("NoiseStackKernel missing param: " + p)
			return 0.0
	var seed: int = int(world_seed) + int(params["seed_offset"])
	var octaves: int = int(params["octaves"])
	var lacunarity: float = float(params["lacunarity"])
	var persistence: float = float(params["persistence"])
	var base_freq: float = float(params["base_frequency_per_m"])
	var base_m: float = float(params["elevation_base_m"])
	var amp_m: float = float(params["elevation_amplitude_m"])

	var total: float = 0.0
	var amp: float = 1.0
	var freq: float = base_freq
	var norm: float = 0.0
	for _i in range(octaves):
		total += _gradient_noise_2d(world_x * freq, world_z * freq, seed) * amp
		norm += amp
		amp *= persistence
		freq *= lacunarity
	if norm > 0.0:
		total /= norm
	return base_m + total * amp_m


func biome_weight(_biome_name: String, _world_x: float, _world_z: float,
                  _world_seed: int, _params: Dictionary) -> float:
	return 1.0
```

```gdscript
class_name KernelComposer
extends RefCounted

# Combines per-biome kernels at world level. Mirrors
# pipeline/kernel_composer.py.

const BIOME_SEED_STRIDE: int = 1000
const DEFAULT_BIOME_TEMPERATURE: float = 0.5


class BiomeBinding:
	var biome_name: String
	var kernel: Kernel
	var params: Dictionary
	var biome_index: int

	func _init(p_name: String, p_kernel: Kernel, p_params: Dictionary,
	           p_index: int) -> void:
		biome_name = p_name
		kernel = p_kernel
		params = p_params
		biome_index = p_index


var _biomes: Array[BiomeBinding] = []
var _biome_scale_m: float = 1024.0
var _temperature: float = DEFAULT_BIOME_TEMPERATURE


# kernel_registry is a Dictionary: kind_string -> Kernel instance.
# Caller builds this dictionary (typically holding one NoiseStackKernel).
func _init(catalog: Dictionary, kernel_registry: Dictionary,
           biome_temperature: float = DEFAULT_BIOME_TEMPERATURE) -> void:
	_biome_scale_m = float(catalog.get("biome_scale_m", 1024.0))
	_temperature = biome_temperature
	var biomes_arr: Array = catalog["biomes"]
	for i in range(biomes_arr.size()):
		var b: Dictionary = biomes_arr[i]
		if not b.has("generator"):
			push_error("biome %s: missing generator block" % b["name"])
			continue
		var gen: Dictionary = b["generator"]
		var kind: String = gen.get("kernel", "")
		if not kernel_registry.has(kind):
			push_error("biome %s: unknown kernel %s" % [b["name"], kind])
			continue
		var kernel: Kernel = kernel_registry[kind]
		var params: Dictionary = gen.get("params", {})
		_biomes.append(BiomeBinding.new(b["name"], kernel, params, i))


func biome_names() -> Array[String]:
	var names: Array[String] = []
	for b in _biomes:
		names.append(b.biome_name)
	return names


func _biome_scalar(biome_index: int, world_x: float, world_z: float,
                   world_seed: int) -> float:
	var seed: int = world_seed + biome_index * BIOME_SEED_STRIDE
	var u: float = world_x / _biome_scale_m
	var v: float = world_z / _biome_scale_m
	return NoiseStackKernel._gradient_noise_2d(u, v, seed)


func sample_biome_weights(world_x: float, world_z: float,
                          world_seed: int) -> Dictionary:
	var scalars: Array[float] = []
	for b in _biomes:
		scalars.append(_biome_scalar(b.biome_index, world_x, world_z, world_seed))
	var mx: float = scalars[0]
	for s in scalars:
		if s > mx:
			mx = s
	var t: float = max(_temperature, 1e-6)
	var exps: Array[float] = []
	var total: float = 0.0
	for s in scalars:
		var e: float = exp((s - mx) / t)
		exps.append(e)
		total += e
	var result: Dictionary = {}
	for i in range(_biomes.size()):
		result[_biomes[i].biome_name] = exps[i] / total
	return result


func sample_height(world_x: float, world_z: float, world_seed: int) -> float:
	var weights: Dictionary = sample_biome_weights(world_x, world_z, world_seed)
	var h: float = 0.0
	for b in _biomes:
		var w: float = weights[b.biome_name]
		if w < 1e-6:
			continue
		h += w * b.kernel.height(world_x, world_z, world_seed, b.params)
	return h
```

```python
"""Debug helper: generate a clipmap ring mesh as an OBJ for visual
inspection in Blender / any 3D viewer. NOT used at runtime — that's
GDScript's job — but useful to verify the donut + skirt math before
porting.

Usage:
    python build_clipmap_mesh_debug.py \\
        --grid-n 256 --grid-step-m 2.0 \\
        --inner-grid-n 0 \\
        --skirt-depth-m 10.0 \\
        --outermost  # adds an outer-edge skirt too
        --out clipmap_ring0.obj
"""
from __future__ import annotations
import argparse
from pathlib import Path


def build_donut_mesh(grid_n: int, grid_step_m: float,
                     inner_grid_n: int,
                     skirt_depth_m: float = 10.0,
                     outermost: bool = False) -> tuple[list, list]:
    """Build a square donut mesh with skirts.

    grid_n: outer grid resolution (vertices per side).
    grid_step_m: meters between adjacent grid vertices.
    inner_grid_n: if > 0, vertices inside a grid_n × grid_n inner
                  square are skipped (this is where the next finer
                  ring renders). Must be a multiple of 2 and less than
                  grid_n. Centered.
    skirt_depth_m: how far the inner-edge skirt drops below the
                  surface (meters, positive). The runtime shader
                  applies displacement to everything; skirt verts
                  are offset by -skirt_depth_m in mesh-local Y so
                  they end up below the displaced surface.
    outermost: if True, also adds an outer-edge skirt (used on the
              outermost ring to hide the world rim).

    Returns (vertices, triangles) where vertices are (x, y, z) in
    ring-local coords (center is origin; surface verts have y=0,
    skirt verts have y=-skirt_depth_m) and triangles are tuples
    of (i0, i1, i2) indices.
    """
    if inner_grid_n > 0 and inner_grid_n % 2 != 0:
        raise ValueError("inner_grid_n must be even (centered)")
    if inner_grid_n >= grid_n:
        raise ValueError("inner_grid_n must be < grid_n")

    half_extent_m = (grid_n - 1) * grid_step_m * 0.5
    inner_half = inner_grid_n * grid_step_m * 0.5

    # Pass 1: main grid (surface y=0).
    indices = [[-1] * grid_n for _ in range(grid_n)]
    verts = []
    for i in range(grid_n):
        for j in range(grid_n):
            x = -half_extent_m + j * grid_step_m
            z = -half_extent_m + i * grid_step_m
            if inner_grid_n > 0:
                if abs(x) < inner_half - 1e-3 and abs(z) < inner_half - 1e-3:
                    continue
            indices[i][j] = len(verts)
            verts.append((x, 0.0, z))

    triangles = []
    for i in range(grid_n - 1):
        for j in range(grid_n - 1):
            v00 = indices[i][j]
            v10 = indices[i][j + 1]
            v01 = indices[i + 1][j]
            v11 = indices[i + 1][j + 1]
            if -1 in (v00, v10, v01, v11):
                continue
            triangles.append((v00, v10, v11))
            triangles.append((v00, v11, v01))

    # Pass 2: inner-edge skirt. Walk the inner hole's perimeter and
    # emit a downward-extruded quad strip there. Only fires if there
    # IS an inner hole.
    if inner_grid_n > 0:
        # The inner hole occupies grid cells where abs(x) < inner_half
        # AND abs(z) < inner_half. Its perimeter on the surface is
        # the ring of verts where exactly one of (abs(x), abs(z)) is
        # ~= inner_half (i.e. the verts that border the hole).
        # We add a parallel "down" vertex for each such surface vert
        # and triangulate the strip.
        inner_perimeter_pairs: list[tuple[int, int]] = []
        # Walk the inner-hole boundary cells. For each surface vert
        # along the inner boundary, add a -skirt_depth_m duplicate.
        # The boundary verts are those that border the hole — i.e.
        # they exist (index != -1) AND at least one of their 4
        # axial neighbors is inside the hole (index == -1 because
        # we skipped it).
        for i in range(grid_n):
            for j in range(grid_n):
                if indices[i][j] == -1:
                    continue
                # Check neighbors; if any axial neighbor is inside
                # the hole (== -1), this is a boundary vert.
                is_boundary = False
                for di, dj in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                    ni, nj = i + di, j + dj
                    if 0 <= ni < grid_n and 0 <= nj < grid_n:
                        if indices[ni][nj] == -1:
                            is_boundary = True
                            break
                if not is_boundary:
                    continue
                # Add the skirt vert.
                surf_vert = verts[indices[i][j]]
                skirt_idx = len(verts)
                verts.append((surf_vert[0], -skirt_depth_m, surf_vert[2]))
                inner_perimeter_pairs.append((indices[i][j], skirt_idx))
        # Triangulate the skirt strip. We need to order the perimeter
        # verts around the hole — simplest: sort by angle from hole
        # center.
        import math
        inner_perimeter_pairs.sort(
            key=lambda p: math.atan2(verts[p[0]][2], verts[p[0]][0])
        )
        for k in range(len(inner_perimeter_pairs)):
            sa, da = inner_perimeter_pairs[k]
            sb, db = inner_perimeter_pairs[(k + 1) % len(inner_perimeter_pairs)]
            # Two tris for the quad (sa, sb, db, da).
            triangles.append((sa, sb, db))
            triangles.append((sa, db, da))

    # Pass 3: outermost-edge skirt (optional).
    if outermost:
        outer_perimeter_pairs: list[tuple[int, int]] = []
        # Border verts are i==0 or i==grid_n-1 or j==0 or j==grid_n-1
        for i in range(grid_n):
            for j in range(grid_n):
                if indices[i][j] == -1:
                    continue
                if not (i == 0 or i == grid_n - 1 or j == 0 or j == grid_n - 1):
                    continue
                surf_vert = verts[indices[i][j]]
                skirt_idx = len(verts)
                verts.append((surf_vert[0], -skirt_depth_m, surf_vert[2]))
                outer_perimeter_pairs.append((indices[i][j], skirt_idx))
        import math
        outer_perimeter_pairs.sort(
            key=lambda p: math.atan2(verts[p[0]][2], verts[p[0]][0])
        )
        for k in range(len(outer_perimeter_pairs)):
            sa, da = outer_perimeter_pairs[k]
            sb, db = outer_perimeter_pairs[(k + 1) % len(outer_perimeter_pairs)]
            # Outer skirts wind opposite direction (faces point outward).
            triangles.append((sa, db, sb))
            triangles.append((sa, da, db))

    return verts, triangles


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--ring", type=int, default=0)
    ap.add_argument("--grid-n", type=int, default=64)
    ap.add_argument("--grid-step-m", type=float, default=8.0)
    ap.add_argument("--inner-grid-n", type=int, default=0)
    ap.add_argument("--skirt-depth-m", type=float, default=10.0)
    ap.add_argument("--outermost", action="store_true",
                    help="Also add an outer-edge skirt.")
    ap.add_argument("--out", required=True)
    args = ap.parse_args()
    verts, tris = build_donut_mesh(args.grid_n, args.grid_step_m,
                                   args.inner_grid_n,
                                   skirt_depth_m=args.skirt_depth_m,
                                   outermost=args.outermost)
    p = Path(args.out)
    with p.open("w", encoding="utf-8") as f:
        for v in verts:
            f.write(f"v {v[0]} {v[1]} {v[2]}\n")
        for t in tris:
            f.write(f"f {t[0]+1} {t[1]+1} {t[2]+1}\n")
    print(f"wrote {len(verts)} verts, {len(tris)} tris to {p}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```gdscript
class_name ClipmapRing
extends Node3D

# One clipmap ring: a square donut mesh that translates with the camera
# at integer multiples of its grid step. Loaded with a heightmap texture
# the vertex shader displaces by.
#
# Geometry parameters (set by ClipmapWorld at instantiation):
#   ring_index:   0..N-1 (0 = innermost, finest)
#   grid_n:       vertices per side (typically 256)
#   grid_step_m:  meters between adjacent grid verts (2, 4, 8, 16...)
#   inner_grid_n: vertices skipped in the center (= next finer ring's grid_n)
#                  0 for the innermost ring (no hole)
#
# Per-frame: the ring's global_position is snapped to (cam_x, 0, cam_z)
# rounded to the nearest multiple of grid_step_m. This keeps vertex
# positions aligned with the world grid as the camera moves.

@export var ring_index: int = 0
@export var grid_n: int = 256
@export var grid_step_m: float = 2.0
@export var inner_grid_n: int = 0
# Skirt depth (meters below surface). See spec
# "Ring boundary stitching (REQUIRED)" — skirt verts hide the
# heightmap-discontinuity crack between adjacent rings.
@export var skirt_depth_m: float = 10.0
# If true, this is the outermost ring; we add an outer-edge skirt
# too to hide the "world rim" behind the bound.
@export var outermost: bool = false

var _mesh_instance: MeshInstance3D
var _mesh: ArrayMesh


func _ready() -> void:
	_mesh = _build_donut_mesh(grid_n, grid_step_m, inner_grid_n,
	                          skirt_depth_m, outermost)
	_mesh_instance = MeshInstance3D.new()
	_mesh_instance.mesh = _mesh
	_mesh_instance.cast_shadow = GeometryInstance3D.SHADOW_CASTING_SETTING_OFF
	add_child(_mesh_instance)


func set_shared_material(mat: Material) -> void:
	if _mesh_instance != null:
		_mesh_instance.material_override = mat


func snap_to_camera(cam_world_xz: Vector2) -> void:
	"""Translate this ring so its center is at the camera position
	snapped to the ring's grid_step_m. Keeps vertex positions aligned
	with the world grid as camera moves."""
	var snap: float = grid_step_m
	var sx: float = roundf(cam_world_xz.x / snap) * snap
	var sz: float = roundf(cam_world_xz.y / snap) * snap
	global_position = Vector3(sx, 0.0, sz)


func get_world_rect_m() -> Rect2:
	"""Returns the world-XZ rect covered by this ring's outer extent
	(useful for picking which ring a sample point belongs to)."""
	var half: float = (float(grid_n) - 1.0) * grid_step_m * 0.5
	var origin: Vector2 = Vector2(global_position.x - half,
	                              global_position.z - half)
	return Rect2(origin, Vector2(half * 2.0, half * 2.0))


# Build the donut mesh. Mirrors pipeline/build_clipmap_mesh_debug.py.
func _build_donut_mesh(p_grid_n: int, p_grid_step_m: float,
                       p_inner_grid_n: int,
                       p_skirt_depth_m: float = 10.0,
                       p_outermost: bool = false) -> ArrayMesh:
	if p_inner_grid_n > 0 and p_inner_grid_n % 2 != 0:
		push_error("inner_grid_n must be even")
	if p_inner_grid_n >= p_grid_n:
		push_error("inner_grid_n must be < grid_n")
	var half_extent: float = (float(p_grid_n) - 1.0) * p_grid_step_m * 0.5
	var inner_half: float = float(p_inner_grid_n) * p_grid_step_m * 0.5

	# Two passes: assign indices skipping hole, then build triangles.
	var indices: Array = []  # [grid_n][grid_n] -> int (-1 if skipped)
	for i in range(p_grid_n):
		var row: Array[int] = []
		for j in range(p_grid_n):
			row.append(-1)
		indices.append(row)
	var positions := PackedVector3Array()
	for i in range(p_grid_n):
		for j in range(p_grid_n):
			var x: float = -half_extent + float(j) * p_grid_step_m
			var z: float = -half_extent + float(i) * p_grid_step_m
			if p_inner_grid_n > 0:
				if abs(x) < inner_half - 1e-3 and abs(z) < inner_half - 1e-3:
					continue
			indices[i][j] = positions.size()
			positions.append(Vector3(x, 0.0, z))

	var idx_buf := PackedInt32Array()
	for i in range(p_grid_n - 1):
		for j in range(p_grid_n - 1):
			var v00: int = indices[i][j]
			var v10: int = indices[i][j + 1]
			var v01: int = indices[i + 1][j]
			var v11: int = indices[i + 1][j + 1]
			if v00 == -1 or v10 == -1 or v01 == -1 or v11 == -1:
				continue
			idx_buf.append(v00); idx_buf.append(v10); idx_buf.append(v11)
			idx_buf.append(v00); idx_buf.append(v11); idx_buf.append(v01)

	# Skirt geometry — hides ring-boundary cracks. See spec
	# "Ring boundary stitching (REQUIRED)". Skirt verts are placed at
	# the inner-hole perimeter (and outer perimeter on the outermost
	# ring) with mesh-local Y = -skirt_depth_m. The vertex shader
	# displaces them with the heightmap the same way it displaces the
	# surface verts, so the skirt sits at (surface_height -
	# skirt_depth_m) at every point of the perimeter.
	if p_inner_grid_n > 0:
		_add_skirt_strip(positions, indices, idx_buf, p_grid_n,
		                  true,   # inner edge
		                  p_skirt_depth_m)
	if p_outermost:
		_add_skirt_strip(positions, indices, idx_buf, p_grid_n,
		                  false,  # outer edge
		                  p_skirt_depth_m)

	# Normals — flat upward; the vertex shader displaces per-vertex
	# and derives surface normals from heightmap finite-differences.
	# So mesh normals are placeholders.
	var normals := PackedVector3Array()
	for _i in range(positions.size()):
		normals.append(Vector3(0, 1, 0))

	# UVs — vertex.xz in mesh-local meters; the shader divides by
	# ring_extent_m to get the ring's [0..1] UV.
	var uvs := PackedVector2Array()
	for v in positions:
		uvs.append(Vector2(v.x, v.z))

	var arrays: Array = []
	arrays.resize(Mesh.ARRAY_MAX)
	arrays[Mesh.ARRAY_VERTEX] = positions
	arrays[Mesh.ARRAY_NORMAL] = normals
	arrays[Mesh.ARRAY_TEX_UV] = uvs
	arrays[Mesh.ARRAY_INDEX] = idx_buf

	var mesh := ArrayMesh.new()
	mesh.add_surface_from_arrays(Mesh.PRIMITIVE_TRIANGLES, arrays)
	return mesh


# Helper: append skirt verts (duplicates of perimeter verts at
# y=-skirt_depth) and the strip triangles into the in-progress
# packed arrays.
#
# inner_edge=true: walks the inner-hole perimeter (boundary verts
#   whose axial neighbor is in the hole). Triangle winding produces
#   outward-facing normals (visible from above the surface).
# inner_edge=false: walks the outer-mesh-edge perimeter (i==0 or
#   i==grid_n-1 or j==0 or j==grid_n-1). Triangle winding produces
#   outward-facing normals (visible from outside the world).
func _add_skirt_strip(positions: PackedVector3Array, indices: Array,
                      idx_buf: PackedInt32Array, p_grid_n: int,
                      inner_edge: bool, skirt_depth: float) -> void:
	var perimeter: Array[Vector2i] = []  # (surface_idx, skirt_idx) pairs as Vector2i
	# Gather perimeter verts in their (surface, skirt-dup) pairs.
	for i in range(p_grid_n):
		for j in range(p_grid_n):
			if indices[i][j] == -1:
				continue
			var is_boundary: bool = false
			if inner_edge:
				for offsets in [Vector2i(-1, 0), Vector2i(1, 0),
				                 Vector2i(0, -1), Vector2i(0, 1)]:
					var ni: int = i + offsets.y
					var nj: int = j + offsets.x
					if ni >= 0 and ni < p_grid_n and nj >= 0 and nj < p_grid_n:
						if indices[ni][nj] == -1:
							is_boundary = true
							break
			else:
				if i == 0 or i == p_grid_n - 1 or j == 0 or j == p_grid_n - 1:
					is_boundary = true
			if not is_boundary:
				continue
			var surf_idx: int = indices[i][j]
			var surf_pos: Vector3 = positions[surf_idx]
			var skirt_idx: int = positions.size()
			positions.append(Vector3(surf_pos.x, -skirt_depth, surf_pos.z))
			perimeter.append(Vector2i(surf_idx, skirt_idx))
	# Sort perimeter by angle around (0, 0) so consecutive pairs form
	# adjacent quads.
	perimeter.sort_custom(func(a: Vector2i, b: Vector2i) -> bool:
		var pa: Vector3 = positions[a.x]
		var pb: Vector3 = positions[b.x]
		return atan2(pa.z, pa.x) < atan2(pb.z, pb.x)
	)
	# Triangulate the strip.
	for k in range(perimeter.size()):
		var sa: int = perimeter[k].x
		var da: int = perimeter[k].y
		var next: int = (k + 1) % perimeter.size()
		var sb: int = perimeter[next].x
		var db: int = perimeter[next].y
		if inner_edge:
			# Wind so the surface faces "out of the hole" (toward camera).
			idx_buf.append(sa); idx_buf.append(sb); idx_buf.append(db)
			idx_buf.append(sa); idx_buf.append(db); idx_buf.append(da)
		else:
			# Outer skirt: face "out of the world."
			idx_buf.append(sa); idx_buf.append(db); idx_buf.append(sb)
			idx_buf.append(sa); idx_buf.append(da); idx_buf.append(db)
```

```gdscript
class_name ClipmapWorld
extends Node3D

# Scene-root runtime for the scale_v2 world. v1 of this script does
# only the clipmap geometry setup; heightmap-from-kernel-composer and
# clipmap splat come in Stages 3+4.

@export var ring_count: int = 4
@export var ring_grid_n: int = 256       # 256 verts per side per ring
@export var ring_grid_step_base_m: float = 2.0  # ring 0 step; ring N step = base * 2^N
@export var debug_material_path: String = ""    # set in scale_v2 debug scene
@export var camera_path: NodePath
@export var update_interval_s: float = 0.05      # how often to snap rings to camera

var _rings: Array[ClipmapRing] = []
var _camera: Camera3D = null
var _update_clock: float = 0.0


func _ready() -> void:
	if camera_path != NodePath(""):
		var node := get_node_or_null(camera_path)
		if node is Camera3D:
			_camera = node
	# Spawn N rings, sharing one debug material.
	var debug_mat: Material = null
	if not debug_material_path.is_empty():
		var res := load(debug_material_path)
		if res is Material:
			debug_mat = res
	# Each ring's inner_grid_n = next-finer ring's grid_n (so it has
	# a hole where the finer ring renders). Innermost ring has no hole.
	for i in range(ring_count):
		var ring := ClipmapRing.new()
		ring.name = "Ring_%d" % i
		ring.ring_index = i
		ring.grid_n = ring_grid_n
		ring.grid_step_m = ring_grid_step_base_m * pow(2.0, float(i))
		# Skirt + outermost flag (see spec
		# "Ring boundary stitching (REQUIRED)" in component 2).
		ring.skirt_depth_m = 10.0
		ring.outermost = (i == ring_count - 1)
		# Inner hole matches the next-finer ring's outer extent in
		# grid units of THIS ring. Inner ring's outer = (grid_n-1) * step_inner.
		# Inner expressed in this ring's step = (grid_n - 1) * step_inner / step_this
		#                                       = (grid_n - 1) / 2 (since step_inner = step_this / 2).
		if i == 0:
			ring.inner_grid_n = 0
		else:
			# This ring's step is 2× the previous ring's step. The
			# previous ring's outer extent is (grid_n - 1) * (step / 2)
			# meters; in this ring's units that's (grid_n - 1) / 2.
			# Round up to even.
			ring.inner_grid_n = (ring_grid_n - 1) / 2 + ((ring_grid_n - 1) / 2) % 2
		add_child(ring)
		_rings.append(ring)
		if debug_mat != null:
			ring.set_shared_material(_per_ring_material(debug_mat, i))


func _process(delta: float) -> void:
	_update_clock += delta
	if _update_clock < update_interval_s:
		return
	_update_clock = 0.0
	if _camera == null:
		return
	var cam_xz: Vector2 = Vector2(_camera.global_position.x,
	                              _camera.global_position.z)
	for r in _rings:
		r.snap_to_camera(cam_xz)


# Make a per-ring material instance with `ring_index` set, sharing the
# rest of the shader params with the global debug material.
func _per_ring_material(base_mat: Material, idx: int) -> Material:
	if not (base_mat is ShaderMaterial):
		return base_mat
	var inst: ShaderMaterial = (base_mat as ShaderMaterial).duplicate(false)
	inst.set_shader_parameter("ring_index", idx)
	return inst
```

```gdscript
		return

		return

	                              _camera.global_position.z)

		var r: ClipmapRing = _rings[i]
		r.snap_to_camera(cam_xz)
		var snap: Vector2 = Vector2(r.global_position.x, r.global_position.z)

			_last_snap_per_ring[i] = snap

# the ring's displacement texture. Also sets the per-ring shader
# uniforms ring_origin_m + ring_extent_m so the vertex shader knows
# the UV-to-world-XZ map.
func _refresh_ring_heightmap(r: ClipmapRing, ring_center: Vector2) -> void:
	var n: int = r.grid_n
	var step: float = r.grid_step_m
	var half_extent: float = (float(n) - 1.0) * step * 0.5
	var origin: Vector2 = ring_center - Vector2(half_extent, half_extent)
	var extent: float = (float(n) - 1.0) * step
	# Allocate float image, sample, fill.
	var img := Image.create(n, n, false, Image.FORMAT_RF)
	for i in range(n):
		for j in range(n):
			var x: float = origin.x + float(j) * step
			var z: float = origin.y + float(i) * step
			var h: float = _composer.sample_height(x, z, world_seed)
			img.set_pixel(j, i, Color(h, 0.0, 0.0, 0.0))
	var tex: ImageTexture = ImageTexture.create_from_image(img)
	r.set_displacement_texture(tex)
	# Per-ring shader uniforms.
	if r.has_method("set_shared_material"):
		# The MeshInstance3D's material_override is the per-ring duplicate
		# we stored in Task 2.4's _per_ring_material. Update its uniforms.
		var mat: Material = r.get_node_or_null(".").get_child(0).material_override if false else null
	# Direct path: walk into the MeshInstance3D under the ring.
	for child in r.get_children():
		if child is MeshInstance3D:
			var mi: MeshInstance3D = child
			if mi.material_override is ShaderMaterial:
				var sm: ShaderMaterial = mi.material_override
				sm.set_shader_parameter("displacement", tex)
				sm.set_shader_parameter("ring_origin_m",
					ring_center - Vector2(half_extent, half_extent))
				sm.set_shader_parameter("ring_extent_m", extent)
				sm.set_shader_parameter("ring_index", r.ring_index)
```

### Task 3.4: Move heightmap generation to `WorkerThreadPool` (async double-buffer)

**Files:**
- Modify: `D:/assets/world 4/the world 4/scripts/ClipmapWorld.gd`

Task 3.3 ships a synchronous `_refresh_ring_heightmap` (~250-500ms
per ring at 256² grid in single-thread GDScript). Walking across
ring boundaries causes visible hitches. The fix is the same
`WorkerThreadPool.add_task` pattern TileTerrain uses today: worker
computes the height array off-thread, main thread creates the
`ImageTexture` + sets shader_parameter on `_finalize_ring_update`.

`RenderingServer` and `PhysicsServer` are main-thread-only in Godot
4.5 — `ImageTexture.create_from_image` MUST run on main. The double-
buffer is: each ring keeps a current displacement texture (the
"live" one bound to the material) + a pending texture being built
by the worker. When the worker finishes, the main thread swaps
pending → current atomically. The shader never sees a half-built
texture.

- [ ] **Step 1: Refactor `_refresh_ring_heightmap` to launch a worker**

Replace the synchronous `_refresh_ring_heightmap` in `ClipmapWorld.gd`
with the async pattern. Add at the top of ClipmapWorld:

```gdscript
# {ring_idx, ring_center, task_id, packed_heights}. _process polls
```

```gdscript

# Kick off async regeneration for a ring. Stores a task entry that
# _poll_ring_tasks (called each _process) checks. Cheap on main
# thread — only allocates the worker task + the result holder.
func _enqueue_ring_refresh(r: ClipmapRing, ring_center: Vector2) -> void:
	var ring_idx: int = r.ring_index
	# Cancel any in-flight task for this ring; the result would
	# correspond to an outdated camera position.
	if _ring_tasks.has(ring_idx):
		# Note: WorkerThreadPool doesn't cancel running tasks. We
		# just mark this slot as superseded; when the old task
		# finishes its result is discarded.
		_ring_tasks[ring_idx]["superseded"] = true
	var task_payload: Dictionary = {
		"ring_idx": ring_idx,
		"ring_center": ring_center,
		"grid_n": r.grid_n,
		"grid_step_m": r.grid_step_m,
	}
	var task_id: int = WorkerThreadPool.add_task(
		_worker_compute_heightmap.bind(task_payload), false, "ring_refresh"
	)
	_ring_tasks[ring_idx] = {
		"task_id": task_id,
		"payload": task_payload,
		"superseded": false,
		"result": null,
	}


# Worker thread function. Computes the heightmap array off the main
# thread. Pure: only reads from _composer + world_seed (both
# read-only on worker side). Writes to a PackedFloat32Array that the
# main thread will pick up via _poll_ring_tasks.
func _worker_compute_heightmap(payload: Dictionary) -> void:
	var n: int = int(payload["grid_n"])
	var step: float = float(payload["grid_step_m"])
	var ring_center: Vector2 = payload["ring_center"]
	var half_extent: float = (float(n) - 1.0) * step * 0.5
	var origin: Vector2 = ring_center - Vector2(half_extent, half_extent)
	var heights := PackedFloat32Array()
	heights.resize(n * n)
	for i in range(n):
		for j in range(n):
			var x: float = origin.x + float(j) * step
			var z: float = origin.y + float(i) * step
			var h: float
			if _in_bounds(x, z):
				h = _composer.sample_height(x, z, world_seed)
			else:
				h = out_of_bound_height_m
			heights[i * n + j] = h
	# Store the result back into the task entry. The main thread reads
	# it on the next _poll_ring_tasks call.
	var ring_idx: int = int(payload["ring_idx"])
	if _ring_tasks.has(ring_idx):
		_ring_tasks[ring_idx]["result"] = heights


# Called from _process. Walks _ring_tasks; for each completed task,
# uploads the heightmap on the main thread + updates the ring.
func _poll_ring_tasks() -> void:
	var done: Array[int] = []
	for ring_idx in _ring_tasks.keys():
		var task: Dictionary = _ring_tasks[ring_idx]
		var task_id: int = int(task["task_id"])
		if not WorkerThreadPool.is_task_completed(task_id):
			continue
		WorkerThreadPool.wait_for_task_completion(task_id)
		done.append(ring_idx)
		if bool(task.get("superseded", false)):
			continue  # discard outdated result
		var result = task.get("result")
		if result == null:
			continue
		# Main thread: build the texture + heightmap collision + bind
		# to the ring's material.
		var heights: PackedFloat32Array = result
		var payload: Dictionary = task["payload"]
		var n: int = int(payload["grid_n"])
		var step: float = float(payload["grid_step_m"])
		var ring_center: Vector2 = payload["ring_center"]
		var r: ClipmapRing = _rings[ring_idx]
		_finalize_ring_upload(r, ring_center, heights, n, step)
	for ring_idx in done:
		_ring_tasks.erase(ring_idx)


# Main thread only. Owns ImageTexture + HeightMapShape3D updates.
func _finalize_ring_upload(r: ClipmapRing, ring_center: Vector2,
                            heights: PackedFloat32Array, n: int,
                            step: float) -> void:
	var half_extent: float = (float(n) - 1.0) * step * 0.5
	var origin: Vector2 = ring_center - Vector2(half_extent, half_extent)
	var extent: float = (float(n) - 1.0) * step
	# Build the displacement texture. R32F = 4 bytes per pixel.
	var img := Image.create_from_data(n, n, false, Image.FORMAT_RF,
	                                   heights.to_byte_array())
	var tex: ImageTexture = ImageTexture.create_from_image(img)
	r.set_displacement_texture(tex)
	# Update the ring's collision shape — see Task 3.5.
	r.update_collision_heightmap(heights, n, step, ring_center)
	# Set per-ring shader uniforms on the per-ring material instance.
	for child in r.get_children():
		if child is MeshInstance3D:
			var mi: MeshInstance3D = child
			if mi.material_override is ShaderMaterial:
				var sm: ShaderMaterial = mi.material_override
				sm.set_shader_parameter("displacement", tex)
				sm.set_shader_parameter("ring_origin_m", origin)
				sm.set_shader_parameter("ring_extent_m", extent)
				sm.set_shader_parameter("ring_index", r.ring_index)
```

### Task 3.5: HeightMapShape3D collision proxy per ring

**Files:**
- Modify: `D:/assets/world 4/the world 4/scripts/ClipmapRing.gd`

GPU-displaced clipmap meshes have flat geometry on the CPU side.
The physics engine sees flat planes — character controllers fall
through the terrain. We add a `HeightMapShape3D` per ring, sized
to match the ring's geometry, and update it whenever the heightmap
texture updates.

The collision data is the same `PackedFloat32Array` the worker
already produced for the displacement texture (Task 3.4's
`_finalize_ring_upload`) — we just hand it to a `CollisionShape3D`
child of the ring.

- [ ] **Step 1: Add CollisionShape3D child + update method to ClipmapRing.gd**

Append to `ClipmapRing.gd`:

```gdscript

# Collision proxy (Stage 3.5).

var _static_body: StaticBody3D = null
var _collision_shape: CollisionShape3D = null
var _height_shape: HeightMapShape3D = null


func _ensure_collision() -> void:
	if _static_body != null:
		return
	_static_body = StaticBody3D.new()
	_static_body.name = "Collision"
	add_child(_static_body)
	_collision_shape = CollisionShape3D.new()
	_static_body.add_child(_collision_shape)
	_height_shape = HeightMapShape3D.new()
	# Initial size — width/depth are vertex counts, NOT meters.
	_height_shape.map_width = grid_n
	_height_shape.map_depth = grid_n
	_collision_shape.shape = _height_shape


func update_collision_heightmap(heights: PackedFloat32Array, n: int,
                                 step: float, ring_center: Vector2) -> void:
	_ensure_collision()
	# HeightMapShape3D's map_data is in mesh-local Y units. The
	# CollisionShape3D's scale is per-vertex spacing — Godot draws
	# the height samples at unit spacing in shape-local space, so we
	# scale the CollisionShape3D by grid_step_m on X+Z to match the
	# mesh.
	_height_shape.map_width = n
	_height_shape.map_depth = n
	_height_shape.map_data = heights
	# Position the collision shape at the ring center, scaled by
	# grid_step_m so vertex N is grid_step_m * N apart.
	_collision_shape.scale = Vector3(step, 1.0, step)
	# Center is global — HeightMapShape3D's origin is the center of
	# the heightfield. We set the shape's transform local to the
	# StaticBody3D (which is local to the ring node, which is at
	# ring_center in world XZ).
	_collision_shape.position = Vector3.ZERO
```

```glsl
// Vertex shader displaces from `displacement` (per-ring R-channel
// heightmap). Fragment shader samples `ring_splat` (per-ring biome
// weight Texture2DArray, N layers = N biomes) + biome PBR arrays
// (standard + hero tiers, same as Axis 6 terrain_world_v2) and
// blends with the Axis 6 within-biome slope blend.

shader_type spatial;
render_mode unshaded, cull_back, depth_draw_opaque;

// Per-biome PBR arrays — same as Axis 6.
uniform sampler2DArray standard_albedo : source_color;
uniform sampler2DArray standard_normal : hint_normal;
uniform sampler2DArray standard_rough;
uniform sampler2DArray standard_ao : hint_default_white;
uniform sampler2DArray hero_albedo : source_color;
uniform sampler2DArray hero_normal : hint_normal;
uniform sampler2DArray hero_rough;
uniform sampler2DArray hero_ao : hint_default_white;

// Per-ring uniforms.
uniform sampler2D displacement : filter_linear, repeat_disable;
uniform sampler2DArray ring_splat : filter_linear, repeat_disable;
uniform vec2 ring_origin_m = vec2(0.0, 0.0);
uniform float ring_extent_m = 512.0;
uniform int ring_index = 0;

// Per-biome (tier, layer) packed for ground/mid/rock — same as v2.
const int MAX_BIOMES = 16;
uniform int biome_ground_packed[MAX_BIOMES];
uniform int biome_mid_packed[MAX_BIOMES];
uniform int biome_rock_packed[MAX_BIOMES];
uniform int num_biomes = 0;

// Slope thresholds (same as v2)
uniform float slope_ground_max : hint_range(0.0, 1.0) = 0.92;
uniform float slope_rock_max   : hint_range(0.0, 1.0) = 0.55;
uniform float world_uv_scale = 0.04;
uniform float albedo_luma_floor : hint_range(0.0, 0.4) = 0.08;
uniform float ao_floor          : hint_range(0.0, 1.0) = 0.72;

// In-shader sun (same as v2)
uniform vec3 sun_dir = vec3(0.30, 0.85, 0.43);
uniform vec3 sun_color : source_color = vec3(1.0, 0.96, 0.88);
uniform float sun_intensity : hint_range(0.0, 3.0) = 1.0;
uniform float lambert_floor : hint_range(0.0, 0.5) = 0.35;
uniform vec3 sky_tint    : source_color = vec3(0.78, 0.84, 0.92);
uniform vec3 ground_tint : source_color = vec3(0.58, 0.60, 0.55);
uniform float ambient_strength : hint_range(0.0, 2.0) = 0.45;

varying vec3 v_world_pos;
varying vec3 v_world_normal;


void vertex() {
	vec3 world_xz = (MODEL_MATRIX * vec4(VERTEX, 1.0)).xyz;
	vec2 uv = (world_xz.xz - ring_origin_m) / ring_extent_m + 0.5;
	float h = texture(displacement, uv).r;
	VERTEX.y = h;

	// Compute normal from displacement texture finite-difference.
	// Stencil = 1 texel in either direction. For accurate world-space
	// normals we'd divide by step_m but for now we approximate.
	float texel_size = 1.0 / float(textureSize(displacement, 0).x);
	float h_x0 = texture(displacement, uv - vec2(texel_size, 0)).r;
	float h_x1 = texture(displacement, uv + vec2(texel_size, 0)).r;
	float h_z0 = texture(displacement, uv - vec2(0, texel_size)).r;
	float h_z1 = texture(displacement, uv + vec2(0, texel_size)).r;
	float dx_m = ring_extent_m * texel_size * 2.0;
	vec3 nx = vec3(-(h_x1 - h_x0) / dx_m, 1.0, 0.0);
	vec3 nz = vec3(0.0, 1.0, -(h_z1 - h_z0) / dx_m);
	vec3 n = normalize(cross(normalize(nz), normalize(nx)));
	NORMAL = n;
	v_world_pos = (MODEL_MATRIX * vec4(VERTEX, 1.0)).xyz;
	v_world_normal = (MODEL_MATRIX * vec4(n, 0.0)).xyz;
}


vec3 luma_floor_v(vec3 color) {
	float lum = dot(color, vec3(0.2126, 0.7152, 0.0722));
	if (lum < albedo_luma_floor) {
		color += vec3(albedo_luma_floor - lum);
	}
	return clamp(color, vec3(0.0), vec3(1.0));
}


void sample_slot(int packed_idx, vec2 uv, out vec3 albedo_out, out float ao_out) {
	if (packed_idx < 0) {
		albedo_out = vec3(0.0);
		ao_out = 1.0;
		return;
	}
	int tier = (packed_idx >> 30) & 1;
	int layer = packed_idx & 0x3FFFFFFF;
	if (tier == 1) {
		albedo_out = luma_floor_v(texture(hero_albedo, vec3(uv, float(layer))).rgb);
		ao_out = texture(hero_ao, vec3(uv, float(layer))).r;
	} else {
		albedo_out = luma_floor_v(texture(standard_albedo, vec3(uv, float(layer))).rgb);
		ao_out = texture(standard_ao, vec3(uv, float(layer))).r;
	}
}


void sample_biome(int biome_idx, vec2 uv, float up,
                  out vec3 albedo_out, out float ao_out) {
	int g_idx = biome_ground_packed[biome_idx];
	int m_idx = biome_mid_packed[biome_idx];
	int r_idx = biome_rock_packed[biome_idx];
	vec3 ga, ma, ra;
	float go, mo, ro;
	sample_slot(g_idx, uv, ga, go);
	sample_slot(m_idx, uv, ma, mo);
	sample_slot(r_idx, uv, ra, ro);
	float ground_w = smoothstep(slope_ground_max - 0.15, slope_ground_max, up);
	float rock_w   = 1.0 - smoothstep(slope_rock_max, slope_rock_max + 0.15, up);
	float mid_w    = max(0.0, 1.0 - ground_w - rock_w);
	float total    = max(ground_w + mid_w + rock_w, 1e-4);
	ground_w /= total; mid_w /= total; rock_w /= total;
	albedo_out = ga * ground_w + ma * mid_w + ra * rock_w;
	ao_out     = go * ground_w + mo * mid_w + ro * rock_w;
}


void fragment() {
	vec2 uv = v_world_pos.xz * world_uv_scale;
	vec3 wn = normalize(v_world_normal);
	float up = clamp(wn.y, 0.0, 1.0);
	vec2 splat_uv = (v_world_pos.xz - ring_origin_m) / ring_extent_m + 0.5;

	vec3 albedo = vec3(0.0);
	float ao = 0.0;
	float wsum = 0.0;
	for (int i = 0; i < MAX_BIOMES; i++) {
		if (i >= num_biomes) {
			break;
		}
		float w = texture(ring_splat, vec3(splat_uv, float(i))).r;
		if (w < 1e-3) {
			continue;
		}
		vec3 a_i;
		float ao_i;
		sample_biome(i, uv, up, a_i, ao_i);
		albedo += a_i * w;
		ao     += ao_i * w;
		wsum   += w;
	}
	if (wsum > 1e-4) {
		albedo /= wsum;
		ao /= wsum;
	}
	ao = clamp(ao, ao_floor, 1.0);

	vec3 sun_n = normalize(sun_dir);
	float ndl = dot(wn, sun_n);
	float lambert = max(ndl, lambert_floor);
	vec3 direct = sun_color * lambert * sun_intensity;
	vec3 ambient_color = mix(ground_tint, sky_tint, up);
	vec3 ambient = ambient_color * ambient_strength;

	vec3 lit = albedo * ao * (direct + ambient);
	lit = luma_floor_v(lit);
	ALBEDO = lit;
}
```

```markdown
- This is the AAA terrain-rendering architecture (Witcher 3, Horizon,
  Hellblade). Building it once means future quality work happens
  inside a sane foundation.
- "Infinite-ready" — `world_bound_m: Vector2.ZERO` disables the
  clamp; no architectural change needed.
- Path 3 kernels are subclasses + registration entries — no
```
